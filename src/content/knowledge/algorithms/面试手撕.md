int find01_binary(const std::string& s) {
    int l = 0, r = s.size() - 1;
    while (l < r) {
        int mid = (l + r ) / 2 ;
        if (s[mid]) == '0' {
            l = mid + 1;
        } else {
            r = mid;
        }
    }

    return l - 1;

## 给你一个 严格递增 的整数数组 rungs ，用于表示梯子上每一台阶的 高度 。当前你正站在高度为 0 的地板上，并打算爬到最后一个台阶。另给你一个整数 dist 。每次移动中，你可以到达下一个距离你当前位置（地板或台阶）不超过 dist 高度的台阶。当然，你也可以在任何正 整数 高度处插入尚不存在的新台阶。返回爬到最后一阶时必须添加到梯子上的 最少 台阶数。

```cpp
题目理解
数组 `rungs` 严格递增，起点高度 0。每次最多向上跳 `dist`。如果相邻两个台阶高度差大于 `dist`，就需要插入新台阶，求**最少插入台阶数量**。

> 公式：两个高度 `prev` 和 `cur`，差值 `delta = cur - prev`
> 需要新增台阶数：`(delta - 1) / dist - 1`，当 `delta > dist` 才需要加。
#include <bits/stdc++.h>
int cal(vector<int>& nums, int dist) {
    int res = 0;
    int cur = 0;
    for (int h : nums) {
        int d = h - cur ;
        res += (d - 1) / dist;
        cur = h ;
    }
    return res ;
}
```

## 二进制数组，最多翻转 1 个 0→1，求最长连续 1 长度（滑动窗口，最多允许窗口内 1 个 0）

    int findMaxConsecutiveOnes(vector<int>& nums) {
        int l = 0;
        int zc = 0;
        int ans = 0;
        for(int r{0}; r < nums.size(); r++) {
            if(nums[r] == 0) zc++;
            while(zc > 1) {
                if (nums[l] == 0) zc--;
                l++;
            }
            ans = max(ans, r - l + 1);
        }
        return ans;
    }
};

## 手撕LRU

```cpp
#include <unordered_map>
struct Node {
    int key;
    int value;
    Node* prev;
    Node* next;
    Node(int k, int v): key(k), value(v), prev(nullptr), next(nullptr) {}
};
class LRUCache {
private:
    std::unordered_map<int, Node*> mp;
    int capacity;
    Node* dummy;

    Node* getNode(int key) {
        auto it = mp.find(key);
        if (it == mp.end()) {
            return nullptr;
        }
        Node* node = it->second;
        remove(node);
        pushFront(node);
        return node;
    }
    void remove(Node* x) {
        x->prev->next = x->next;
        x->next->prev = x->prev;
    }
    void pushFront(Node* x) {
        x->prev = dummy;
        x->next = dummy->next;
        x->prev->next = x;
        x->next->prev = x;
    }
public:
    LRUCache(LRUCache&) = delete;
    LRUCache& operator=(const LRUCache&) = delete;
    LRUCache(int cap) : capacity(cap) {
        dummy = new Node(0, 0);
        dummy->prev = dummy;
        dummy->next = dummy;
    }
    int  Get(int key) {
        auto* node = getNode(key);
        if (!node) return -1;
        return node->value;
    }
    void Put(int key, int value )  {
        auto* node = getNode(key);
        if (node != nullptr) {
            node->value = value;
            return ;
        }
        node = new Node(key, value);
        mp[key] = node;
        pushFront(node);
        if(mp.size() > capacity) {
            auto* backnode = dummy->prev;
            mp.erase(backnode->key);
            remove(backnode);
            delete backnode;
        }
    }
    ~LRUCache() {
        auto *cur = dummy->next;
        while(cur != dummy) {
            auto* del = cur;
            cur = cur->next;
            delete del;
        }
        delete dummy;
    }
};
```

class Solution{
public:
    int maxArea(vector<int>& height) {
        int r = height.size() - 1;
        int l{0}, ans{0};
        while(l < r) {
            // r - l
            ans = max(ans, min(height[l], height[r]) * (r - l));
            if (height[l] < height[r]) l++;
            else r--;
        }
        return ans;
    }

        for (auto& x : nums) {

# 题目：允许一次区间反转的最长连续target

> 二进制数组，**最多可以反转1次连续子区间（区间内0↔1，0变1、1变0，也可以选择不反转）**，求反转后，数组中最长连续等于 `target` 的子数组长度。

暴力思路：枚举所有`[L,R]`区间模拟翻转，再扫描数组统计最长target，复杂度$O(n^2)$，大数据超时。最优解：**Kadane最大子数组，O(n)时间、O(1)空间**。

## 核心转化思路

翻转一段区间`[L,R]`，对target来说，每个位置有收益/损失：

- 若`nums[i] == target`：本来就是目标值，一旦被翻转 → 不再是target，收益 = `-1`
- 若`nums[i] != target`：本来不是目标值，一旦被翻转 → 变成target，收益 = `+1`

设收益数组 $a[i]$：

$$
a[i]=
\begin{cases}
-1 & nums[i]==target \\
1 & nums[i]!=target
\end{cases}
$$

翻转区间$[L,R]$的总收益 = $\sum_{i=L}^{R}a[i]$。

- 总和>0：翻转这段区间，整体可以增加target数量；
- 总和≤0：翻转没有收益，选择**不翻转**。

解题分为两步：

1. **base**：不做任何翻转，原数组中最长连续target的长度。
2. **maxGain**：使用Kadane算法求收益数组`a`的**最大子数组和**（翻转某一段区间能拿到的最大净收益）。

最终答案：
$$\boldsymbol{ans = base + \max(maxGain,\ 0)}$$

> 如果`maxGain ≤ 0`，取0，等价于不翻转任何区间。

## Kadane算法原理

维护两个变量：

- `dp`：**以当前下标结尾的最大子数组和**
  $$dp = \max(a[i],\ dp+a[i])$$
  含义：要么从当前元素新开一段区间；要么把当前元素接在前面的子数组后面。
- `maxGain`：全局记录遍历过程中`dp`的最大值。

## C++ 代码（面试手撕版）

```cpp
#include <vector>
#include <algorithm>
using namespace std;

class Solution {
public:
    int findMax(vector<int>& nums, int target) {
        // 1. base：不翻转，原数组最长连续target
        int base = 0;
        int cur = 0;
        for(int x : nums) {
            if(x == target) {
                cur++;                // 当前这个数就是target，连续长度+1
                base = max(base, cur);// 更新全局最长
            } else {
                cur = 0;              // 当前数不是target，连续断了！重置当前连续计数为0
            }

        }

        // 2. Kadane求收益数组最大子数组和，得到maxGain
        int pre = 0, Max = 0;
        for (int x : nums) {
            int val = (x == target) ? -1 : 1;
            pre = max(pre + val, val);
            Max = max(Max, pre);
        }
        return base + Max;
    }
};
```

## 把石头分成两个子集，求两组和的最小差值。**不限制两组元素数量，只要求所有元素必须分配**。

> 总和 \(total = sum(nums)\)
> 子集 A 和为 S，子集 B 和为 \(total - S\)
> 差值 \(= |total - S - S| = |total - 2S|\)
> 目标：找到**不超过 total/2 的最大 S**，代入得到最小差值。

```cpp
//dp[j] 若干石头， 凑出总和等于j
//背包容量 total / 2
//initialize: dp[0] = true
//从target往下找最大的j, 满足dp[j] = true
//ans = total - 2 * j
#include <vector>
#include <numeric>
using namespace std;
int minSubSetDiff(vector<int>& ss) {
    int total = accumulate(ss.begin(), ss.end(), 0);
    int target = total / 2;
    vector<bool> dp(target + 1, false);
    dp[0] = true;

    for(int x : ss) {
        //01背包， 倒序
        for (int j = target; j >= x; j--) {
            dp[j] = dp[j] || dp[j - x];
        }
    }
    int b = 0;
    for(int j = target; j >= 0; j--) {
        if (dp[j]) {
            b = j;
            break;
        }
    }
    return total - 2 * b;
}
```

## 给定一个二叉树，如何求树中两个节点的最近公共祖先 (LCA)

> 最近公共祖先：两个节点 p、q，深度最大、同时是 p 和 q 祖先的节点。

**递归后序解法**
函数返回值含义：返回当前子树中找到的 p 或者 q；找不到返回 nullptr。

1. 如果 root 为空，或者 root 等于 p 或者 root 等于 q，直接返回 root；
2. 递归左子树得到 left，递归右子树得到 right；
3. - left 非空 && right 非空：说明 p、q 分别分布在左右子树，**root 就是 LCA**，返回 root；
   - left 不为空，right 为空：返回 left（p/q 都在左子树）
   - right 不为空，left 为空：返回 right（p/q 都在右子树）

```
TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
    if(!root || root == p || root == q) return root;
    TreeNode* left = lowestCommonAncestor(root->left,p,q);
    TreeNode* right = lowestCommonAncestor(root->right,p,q);
    if(left && right) return root;
    return left ? left : right;
}
```

## 一个 m*n 的矩阵，从左上角走到右下角，每次只能向右或向下走，如何计算有多少种不同的路径？

> LeetCode 62 不同路径。只能向右、向下。

### 组合数学思路（最优）

一共要向下走 \(m‑1\) 步，向右走 \(n‑1\) 步；总步数 \((m‑1)+(n‑1)=m+n‑2\)。
总路径数等价于总步数中选 \(m‑1\) 步向下（剩余向右）：

\(C_{m+n‑2}^{m‑1}\)

### DP 思路

dp [i][j]：走到 i,j 位置的路径数。

- 第一行所有点：只能从左边过来，`dp[0][j]=1`
- 第一列所有点：只能从上边过来，`dp[i][0]=1`
  转移：`dp[i][j] = dp[i‑1][j] + dp[i][j‑1]`；上面来 + 左边来。
  可以滚动数组压缩空间为一维。

时间\(O(mn)\)，空间可以压到\(O(n)\)。
