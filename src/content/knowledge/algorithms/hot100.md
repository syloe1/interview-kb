# 算法题

[1.两数之和](https://leetcode.cn/problems/two-sum/)
> x + y = target, 用map存
```Go
func twoSum(nums []int, target int) []int {
    mp := map[int]int{}
    for r, x := range nums {
        if p, ok := mp[target-x]; ok {
            return []int{r, p}
        }
        mp[x] = r 
    }
    return []int{}
}
```
[2. 前k个高频元素](https://leetcode.cn/problems/top-k-frequent-elements/)
```Go
func topKFrequent(nums []int, k int) []int {
    cnt := map[int]int{} 
    //buckets[频率] = []元素
    /*
        1 3 cnt[1] = 3 
        2 2 cnt[2] = 2
        3 1 cnt[3] = 1
        mC = 2 
        _ _  _  _
    */
    mC := 0 
    // 桶的标号是出现频率
    //倒序枚举桶， 最先碰到就是最高频率， 取出桶里的val
    // 0 ..... .. Mc
    for _, x := range nums {
        cnt[x]++
        mC = max(mC, cnt[x])
    }
    // bucket[i]是出现次数 ， bucket[i] val是值 bucker[3] = 1 1出现3次
    // 可能元素出现次数相同， 所以使用二维切片
    buckets := make([][]int, mC + 1)
    // 1 3  bucket[3] = 1
    for x, c := range cnt {
        buckets[c] = append(buckets[c], x)
    }
    ans := make([]int,0,k)
    for i := mC; len(ans) < k; i-- {
        //把 buckets[i] 这个切片里的所有元素，一个一个追加到 ans 里面。
        ans = append(ans, buckets[i]...)
    }
    return ans 
}
```
